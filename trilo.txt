n queens with bitmask optimization
#include <iostream>
using namespace std;

int N = 8;  // board size
long long solutionCount = 0;

// Solve using bitmask optimization
void solveNQueensBit(int row, unsigned long long cols, unsigned long long d1, unsigned long long d2) {  // cols, d1, d2 track columns and two diagonals
    if (row == N) {
        solutionCount++;
        return;
    }
    // Compute free positions: bits = 1 for free column positions in this row
    unsigned long long free = ((1ULL << N) - 1ULL) & ~(cols | d1 | d2);  // bitmask of all free positions in this row
    while (free) {
        unsigned long long bit = free & -free;  // lowest set bit
        free -= bit;                            // remove this bit from free
        // Place queen at this bit position and move to next row
        solveNQueensBit(row + 1, cols | bit, (d1 | bit) << 1, (d2 | bit) >> 1);
    }
}

int main() {
    solutionCount = 0;
    solveNQueensBit(0, 0ULL, 0ULL, 0ULL);
    cout << "Number of solutions for " << N << "-Queens = " << solutionCount << endl;
}

const int MAXN = 15;
int N = 4;  // using 4 for demonstration (2 solutions)
bool col[MAXN], diag1[2*MAXN], diag2[2*MAXN];
int queenPos[MAXN];  // queenPos[r] = column index of queen in row r

// Recursively place queens and print all solutions
void solveAllQueens(int row) {
    if (row == N) {
        // Reached a solution, print the board
        for (int r = 0; r < N; ++r) {
            for (int c = 0; c < N; ++c) {
                cout << (queenPos[r] == c ? "Q " : ". ");
            }
            cout << "\n";
        }
        cout << "\n";         // blank line after each solution
        return;
    }
    for (int c = 0; c < N; ++c) {
        if (!col[c] && !diag1[row+c] && !diag2[row-c+N]) {
            // Place queen at (row, c)
            col[c] = diag1[row+c] = diag2[row-c+N] = true;
            queenPos[row] = c;
            solveAllQueens(row+1);
            // Remove queen (backtrack)
            col[c] = diag1[row+c] = diag2[row-c+N] = false;
        }
    }
}

// Backtracking to count all solutions
void countNQueens(int row) {  // place queens row by row
    if (row == N) {
        solutionCount++;
        return;
    }
    // Try placing a queen in each column of this row
    for (int c = 0; c < N; ++c) {
        if (!col[c] && !diag1[row+c] && !diag2[row-c+N]) {
            col[c] = diag1[row+c] = diag2[row-c+N] = true;  // mark (row,c) as occupied
            countNQueens(row+1);
            col[c] = diag1[row+c] = diag2[row-c+N] = false;  // unmark (row,c)
        }
    }
}

z algorithm for pattern matching
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// Compute Z-array for string s
void computeZ(const string& s, vector<int>& Z) { // compute Z array
    int n = s.length();
    Z.assign(n, 0);
    int L = 0, R = 0;
    for (int i = 1; i < n; ++i) {
        if (i > R) {
            L = R = i;
            while (R < n && s[R - L] == s[R]) {
                R++;
            }
            Z[i] = R - L;
            R--;
        } else {
            int k = i - L;
            if (Z[k] < R - i + 1) {
                Z[i] = Z[k];
            } else {
                L = i;
                while (R < n && s[R - L] == s[R]) {
                    R++;
                }
                Z[i] = R - L;
                R--;
            }
        }
    }
}

// Z-algorithm search for pattern in text
void Zsearch(const string& text, const string& pat) { // search using Z algorithm
    string concat = pat + "$" + text;
    int n = concat.length();
    int m = pat.length();
    vector<int> Z;
    computeZ(concat, Z);
    for (int i = m + 1; i < n; ++i) {
        if (Z[i] == m) {
            cout << "Pattern found at index " << i - m - 1 << endl;
        }
    }
}

int main() {
    string text = "abcxxabc";
    string pat = "abc";
    Zsearch(text, pat);
}

kmp algorithm
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// Compute longest prefix-suffix (LPS) array for pattern (KMP preprocessing)
void computeLPS(const string& pat, vector<int>& lps) { // compute LPS array
    int m = pat.length();
    lps.assign(m, 0);
    int len = 0;
    for (int i = 1; i < m; ) {
        if (pat[i] == pat[len]) {
            lps[i++] = ++len;
        } else if (len != 0) {
            len = lps[len-1];
        } else {
            lps[i++] = 0;
        }
    }
}

// KMP search using the LPS array
void KMPsearch(const string& text, const string& pat) { // search using LPS array
    int n = text.length();
    int m = pat.length();
    if (m == 0) return;
    vector<int> lps;
    computeLPS(pat, lps);
    int i = 0, j = 0;
    while (i < n) {
        if (text[i] == pat[j]) {
            i++; j++;
        }
        if (j == m) {
            cout << "Pattern found at index " << i - j << endl;
            j = lps[j - 1];
        } else if (i < n && text[i] != pat[j]) {
            if (j != 0) {
                j = lps[j - 1];
            } else {
                i++;
            }
        }
    }
}

int main() {
    string text = "ABABDABACDABABCABAB";
    string pat = "ABAB";
    KMPsearch(text, pat);
}

rabin karp
#include <iostream>
#include <string>
using namespace std;

// Rabin-Karp algorithm for string matching
void rabinKarpSearch(const string& text, const string& pat) {
    int n = text.length();
    int m = pat.length();
    if (m == 0) return;
    const int d = 256;
    const int q = 101;  // prime modulus for hashing
    int patHash = 0, txtHash = 0;
    int h = 1;
    // Precompute h = d^(m-1) % q
    for (int i = 0; i < m - 1; ++i) h = (h * d) % q;
    // Compute initial hash values for pattern and first window
    for (int i = 0; i < m; ++i) {
        patHash = (patHash * d + pat[i]) % q;
        txtHash = (txtHash * d + text[i]) % q;
    }
    // Slide the pattern over text
    for (int i = 0; i <= n - m; ++i) {
        // If hash values match, verify characters
        if (patHash == txtHash) {
            int j = 0;
            while (j < m && text[i+j] == pat[j]) j++;
            if (j == m) cout << "Pattern found at index " << i << endl;
        }
        // Compute hash of next window
        if (i < n - m) {
            txtHash = (d * (txtHash - text[i] * h) + text[i + m]) % q;
            if (txtHash < 0) txtHash += q;
        }
    }
}

int main() {
    string text = "GEEKS FOR GEEKS";
    string pat = "GEEK";
    rabinKarpSearch(text, pat);
}

int cutRodRec(int n) {
    if (n == 0) return 0;
    int maxVal = INT_MIN;
    for (int i = 1; i <= n; ++i) {
        int val = price[i] + cutRodRec(n - i);

    for (int j = 1; j <= n; ++j) {
        int maxVal = INT_MIN;
        for (int i = 1; i <= j; ++i) {
            if (price[i] + dp[j - i] > maxVal) {
                maxVal = price[i] + dp[j - i];
            }
        }
        dp[j] = maxVal;
    }
    return dp[n];
}

int price[] = {0, 1, 5, 8, 9, 10, 17, 17, 20};
int N = 8;
int cost = 2;  // cost per cut
int dp[100];   // memoization array

// Top-down DP (memoized recursion) with cut cost
int cutRodCost(int n) {
    if (n == 0) return 0;
    if (dp[n] != -1) return dp[n];
    // Option: no cut at all (take full piece)
    int maxVal = price[n];
    // Try every first cut position
    for (int i = 1; i < n; ++i) {
        int total = cutRodCost(i) + cutRodCost(n - i) - cost;
        if (total > maxVal) {
            maxVal = total;
        }
    }
    dp[n] = maxVal;
    return maxVal;
}

int main() {
    memset(dp, -1, sizeof(dp));
    cout << "Maximum value with cut cost " << cost << " for length " << N 
         << " is " << cutRodCost(N) << endl;
}

// Bottom-up DP to maximize number of pieces
int maxPieces(int n) {
    int dp[100];
    dp[0] = 0;
    for (int j = 1; j <= n; ++j) {
        int maxCount = 1; // at least one piece (itself)
        for (int i = 1; i < j; ++i) {
            maxCount = max(maxCount, dp[i] + dp[j - i]);
        }
        dp[j] = maxCount;
    }
    return dp[n];
}

int dims[] = {40, 20, 30, 10, 30};
int n = 4;
int dp[101][101];
int splitIdx[101][101];

// Bottom-up DP to compute optimal parenthesization
void matrixChainOrder() {
    for (int i = 1; i <= n; ++i) {
        dp[i][i] = 0;
    }
    for (int L = 2; L <= n; ++L) {
        for (int i = 1; i <= n - L + 1; ++i) {
            int j = i + L - 1;
            dp[i][j] = INT_MAX;
            for (int k = i; k < j; ++k) {
                int cost = dp[i][k] + dp[k + 1][j] + dims[i - 1] * dims[k] * dims[j];
                if (cost < dp[i][j]) {
                    dp[i][j] = cost;
                    splitIdx[i][j] = k;
                }
            }
        }
    }
}

        cout << " x ";
        printOptimal(splitIdx[i][j] + 1, j);
        cout << ")";
    }
}

// Bottom-up DP for matrix chain multiplication
int matrixChainDP(int N) {
    int dp[101][101];
    // cost is zero for single matrix
    for (int i = 1; i <= N; ++i) {
        dp[i][i] = 0;
    }
    // L is chain length
    for (int L = 2; L <= N; ++L) {
        for (int i = 1; i <= N - L + 1; ++i) {
            int j = i + L - 1;
            dp[i][j] = INT_MAX;
            for (int k = i; k < j; ++k) {
                int cost = dp[i][k] + dp[k + 1][j] + dims[i - 1] * dims[k] * dims[j];
                if (cost < dp[i][j]) {
                    dp[i][j] = cost;
                }
            }
        }
    }
    return dp[1][N];
}

// Top-down memoized DP for matrix chain
int matrixChainMemo(int i, int j) {
    if (i == j) return 0;
    if (dp[i][j] != -1) return dp[i][j];
    dp[i][j] = INT_MAX;
    for (int k = i; k < j; ++k) {
        int cost = matrixChainMemo(i, k) + matrixChainMemo(k + 1, j)
                   + dims[i - 1] * dims[k] * dims[j];
        if (cost < dp[i][j]) {
            dp[i][j] = cost;
        }
    }
    return dp[i][j];
}

        for (int b = 0; b < 100; ++b)
            dp[a][b] = -1;

         << matrixChainMemo(1, n) << endl;

#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

string X = "ABCBDAB";
string Y = "BDCABA";
int dp[101][101];

int main() {
    int m = X.size();
    int n = Y.size();
    // Compute LCS length DP table
    for (int i = 0; i <= m; ++i) {
        for (int j = 0; j <= n; ++j) {
            if (i == 0 || j == 0) {
                dp[i][j] = 0;
            } else if (X[i - 1] == Y[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    // Backtrack to find one LCS
    int i = m, j = n;
    string lcs = "";
    while (i > 0 && j > 0) {
        if (X[i - 1] == Y[j - 1]) {
            lcs.push_back(X[i - 1]);
            i--; j--;
        } else if (dp[i - 1][j] >= dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }
    reverse(lcs.begin(), lcs.end());
    cout << "One LCS: " << lcs << endl;
}

// Function to compute all LCS sequences using dp table
set<string> lcsAll(int i, int j) {
    if (i == 0 || j == 0) {
        return {""};
    }
    if (X[i-1] == Y[j-1]) {
        set<string> prev = lcsAll(i-1, j-1);
        set<string> res;
        for (auto s : prev) {
            res.insert(s + X[i-1]);
        }
        return res;
    } else {
        set<string> res;
        if (dp[i-1][j] == dp[i][j]) {
            res = lcsAll(i-1, j);
        }
        if (dp[i][j-1] == dp[i][j]) {
            set<string> res2 = lcsAll(i, j-1);
            res.insert(res2.begin(), res2.end());
        }
        return res;
    }
}

greedy activity selector
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Activity { int start, finish; };

int main() {
    // Example activities (start, finish)
    vector<Activity> activities = {
        {1, 4}, {3, 5}, {0, 6}, {5, 7}, {3, 8},
        {5, 9}, {6, 10}, {8, 11}, {8, 12}, {2, 13}, {12, 14}
    };
    // Sort by finish time
    sort(activities.begin(), activities.end(), [](const Activity& a, const Activity& b) {
        return a.finish < b.finish;
    });
    // Select activities greedily
    vector<Activity> selected;
    selected.push_back(activities[0]);
    for (size_t i = 1; i < activities.size(); ++i) {
        if (activities[i].start >= selected.back().finish) {
            selected.push_back(activities[i]);
        }
    }
    // Print selected activities
    cout << "Selected activities (start, finish): ";
    for (auto act : selected) {
        cout << "(" << act.start << "," << act.finish << ") ";
    }
    cout << endl;
}

weighted activity selection
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Job { int start, finish, profit; };

int main() {
    vector<Job> jobs = {
        {2, 5, 6}, {1, 3, 5}, {4, 6, 5}, {6, 7, 4}
    };
    // Sort jobs by finish time
    sort(jobs.begin(), jobs.end(), [](const Job& a, const Job& b) {
        return a.finish < b.finish;
    });
    int n = jobs.size();
    // Compute p (last non-conflicting job index) for each job
    vector<int> p(n);
    for (int i = 0; i < n; ++i) {
        p[i] = -1;
        for (int j = i - 1; j >= 0; --j) {
            if (jobs[j].finish <= jobs[i].start) {
                p[i] = j;
                break;
            }
        }
    }
    // DP to compute max profit
    vector<int> dp(n);
    dp[0] = jobs[0].profit;
    for (int i = 1; i < n; ++i) {
        int includeProfit = jobs[i].profit;
        if (p[i] != -1) includeProfit += dp[p[i]];
        dp[i] = (includeProfit > dp[i - 1] ? includeProfit : dp[i - 1]);
    }
    cout << "Maximum profit = " << dp[n - 1] << endl;
}

#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
using namespace std;

struct Task { int duration, deadline; };

int main() {
    vector<Task> tasks = {
        {5, 5}, {1, 6}, {2, 7}, {2, 8}, {4, 9}
    };
    // Sort tasks by increasing deadline
    sort(tasks.begin(), tasks.end(), [](const Task& a, const Task& b) {
        return a.deadline < b.deadline;
    });
    priority_queue<int> pq;
    int currentTime = 0;
    // Use a max-heap to remove longest task if needed
    for (auto& task : tasks) {
        currentTime += task.duration;
        pq.push(task.duration);
        if (currentTime > task.deadline) {
            currentTime -= pq.top();  // remove longest task
            pq.pop();
        }
    }
    cout << "Maximum number of tasks = " << pq.size() << endl;
}

bool match(string text, string pattern) { int n = text.length(), m = pattern.length(); vector<vector<bool>> dp(n+1, vector<bool>(m+1, false)); dp[0][0] = true; for (int j = 1; j <= m && pattern[j-1] == ''; ++j) dp[0][j] = true; for (int i = 1; i <= n; ++i) for (int j = 1; j <= m; ++j) if (pattern[j-1] == '' || pattern[j-1] == text[i-1]) dp[i][j] = dp[i-1][j] || (pattern[j-1] != '*' ? dp[i-1][j-1] : dp[i][j-1]); else if (pattern[j-1] == '?') dp[i][j] = dp[i-1][j-1]; return dp[n][m]; }

int main() { string text = "abcde", pattern = "a*?e"; cout << (match(text, pattern) ? "Match\n" : "No Match\n"); }

Matrix Chain Multiplication — Count Parenthesizations

#include <iostream> #include <vector> using namespace std;

int countWays(int n) { vector<vector<int>> dp(n, vector<int>(n, 0)); for (int l = 2; l < n; ++l) for (int i = 1; i < n - l + 1; ++i) { int j = i + l - 1; for (int k = i; k < j; ++k) dp[i][j] += (dp[i][k] ? dp[i][k] : 1) * (dp[k+1][j] ? dp[k+1][j] : 1); } return dp[1][n-1]; }

int main() { int n = 5; // A1 A2 A3 A4 => 4 matrices cout << "Parenthesization count: " << countWays(n) << endl; }

 LCS of Three Strings

#include <iostream> #include <vector> #include <string> using namespace std;

int LCS3(string a, string b, string c) { int n = a.size(), m = b.size(), o = c.size(); vector<vector<vector<int>>> dp(n+1, vector<vector<int>>(m+1, vector<int>(o+1, 0))); for (int i = 1; i <= n; ++i) for (int j = 1; j <= m; ++j) for (int k = 1; k <= o; ++k) if (a[i-1] == b[j-1] && b[j-1] == c[k-1]) dp[i][j][k] = dp[i-1][j-1][k-1] + 1; else dp[i][j][k] = max({dp[i-1][j][k], dp[i][j-1][k], dp[i][j][k-1]}); return dp[n][m][o]; }

int main() { cout << "LCS length: " << LCS3("abc", "ac", "bcc") << endl; }

Activity Selection with Rest Time

#include <iostream> #include <vector> #include <algorithm> using namespace std;

N-Queens on a Toroidal Board

A toroidal board wraps around horizontally and vertically. Queens can attack “around the edges” too.

#include <iostream> #include <vector> using namespace std;

bool isSafe(const vector<int>& board, int row, int col, int n) { for (int i = 0; i < row; ++i) { int c = board[i]; if (c == col) return false; if ((i + c) % n == (row + col) % n || (i - c + n) % n == (row - col + n) % n) return false; } return true; }

void solve(vector<int>& board, int row, int n, int& count) { if (row == n) { ++count; return; } for (int col = 0; col < n; ++col) { if (isSafe(board, row, col, n)) { board[row] = col; solve(board, row + 1, n, count); } } }

int main() { int n = 5; // No toroidal solution for N=6 vector<int> board(n); int count = 0; solve(board, 0, n, count); cout << "Toroidal N-Queens solutions: " << count << endl; }

Rod Cutting with Limited Stock per Length

Only allowed to use a piece length a limited number of times.

#include <iostream> #include <vector> using namespace std;

int rodCutStock(int n, vector<int> price, vector<int> stock) { vector<int> dp(n+1, 0); for (int len = 1; len <= n; ++len) { for (int s = 0; s < stock[len]; ++s) { for (int j = n; j >= len; --j) dp[j] = max(dp[j], dp[j - len] + price[len]); } } return dp[n]; }

Matrix Chain with Memory Constraint

Only 2 matrices can be in memory at once. Simulate optimal parenthesis under memory limits.

#include <iostream> #include <vector> using namespace std;

int matrixChainMemoLimit(vector<int>& dims) { int n = dims.size() - 1; vector<vector<int>> dp(n, vector<int>(n, 1e9)); for (int i = 0; i < n; ++i) dp[i][i] = 0; for (int len = 2; len <= n; ++len) for (int i = 0; i + len - 1 < n; ++i) { int j = i + len - 1; for (int k = i; k < j; ++k) { // Simulate memory limit by disallowing adjacent chains of size > 2 if (k - i + 1 > 2 || j - k > 2) continue; int cost = dp[i][k] + dp[k+1][j] + dims[i]*dims[k+1]*dims[j+1]; dp[i][j] = min(dp[i][j], cost); } } return dp[0][n-1]; }

int main() { vector<int> dims = {10, 30, 5, 60}; cout << "Min cost under memory limit: " << matrixChainMemoLimit(dims) << endl; }

Matrix Chain Variation with Mid-Three Subchains

#include <iostream> #include <vector> #include <climits> using namespace std; 

int matrixChainOrder(const vector<int>& p, int n) { vector<vector<int>> dp(n, vector<int>(n, 0)); for (int len = 2; len < n; len++) { for (int i = 0; i < n - len; i++) { int j = i + len; dp[i][j] = INT_MAX; for (int k = i; k < j; k++) { dp[i][j] = min(dp[i][j], dp[i][k] + dp[k + 1][j] + p[i] * p[k + 1] * p[j]); } } } return dp[0][n - 1]; } 

int midThreeSubchainMultiply(const vector<int>& p, int n) { int mid = (n - 1) / 2; vector<int> firstChain(p.begin(), p.begin() + mid); int cost1 = matrixChainOrder(firstChain, mid); vector<int> secondChain(p.begin() + mid - 1, p.begin() + mid + 2); int cost2 = matrixChainOrder(secondChain, 3); vector<int> thirdChain(p.begin() + mid + 2, p.end()); int cost3 = matrixChainOrder(thirdChain, n - mid - 2); int mergeCost = p[mid - 1] * p[mid] * p[mid + 1]; return cost1 + cost2 + cost3 + mergeCost; }

N-Queens with Minimum Queens to Cover All Cells (Dominating Set)

Place minimum number of queens such that every cell is attacked or occupied. (Brute-force for small n)

#include <iostream> #include <vector> #include <cmath> using namespace std;

int n = 4, minQueens = 1e9; vector<vector<int>> board;

bool isCovered(vector<vector<int>>& b) { vector<vector<bool>> covered(n, vector<bool>(n, false)); for (int i = 0; i < n; ++i) for (int j = 0; j < n; ++j) if (b[i][j]) for (int dx = -n; dx <= n; ++dx) for (int dy = -n; dy <= n; ++dy) if (abs(dx) == abs(dy) || dx == 0 || dy == 0) if (i+dx >= 0 && i+dx < n && j+dy >= 0 && j+dy < n) covered[i+dx][j+dy] = true; for (int i = 0; i < n; ++i) for (int j = 0; j < n; ++j) if (!covered[i][j]) return false; return true; }

void dfs(int r, int c, int placed) { if (placed >= minQueens) return; if (r == n) { if (isCovered(board)) minQueens = placed; return; } int nr = r + (c+1)/n, nc = (c+1)%n; board[r][c] = 1; dfs(nr, nc, placed+1); board[r][c] = 0; dfs(nr, nc, placed); }

int main() { board = vector<vector<int>>(n, vector<int>(n, 0)); dfs(0, 0, 0); cout << "Minimum queens needed: " << minQueens << endl; }


Rabin-Karp with Multiple Patterns

#include <iostream> #include <unordered_set> #include <vector> using namespace std;

const int BASE = 256, MOD = 1e9+7;

int getHash(string s) { long h = 0; for (char c : s) h = (h * BASE + c) % MOD; return h; }

void matchPatterns(string text, vector<string>& patterns) { int m = patterns[0].length(); long h = 0, power = 1; for (int i = 0; i < m-1; ++i) power = (power * BASE) % MOD;

unordered_set<int> hashes;
for (string p : patterns)
    hashes.insert(getHash(p));

for (int i = 0; i < text.length(); ++i) {
    if (i < m) h = (h * BASE + text[i]) % MOD;
    else {
        h = (h - text[i - m] * power % MOD + MOD) % MOD;
        h = (h * BASE + text[i]) % MOD;
    }
    if (i >= m-1 && hashes.count(h))
        cout << "Match at index " << i - m + 1 << endl;
}
}

int main() { string text = "abcxabcdabxabcdabcdabcy"; vector<string> patterns = {"abcdabcy", "abcx"}; matchPatterns(text, patterns); }

Rod Cutting with Quality Penalty

Each cut reduces total quality by 1 point. Maximize net value.

#include <iostream> #include <vector> using namespace std;

int rodCutPenalty(int n, vector<int>& price) { vector<int> dp(n+1, 0); for (int i = 1; i <= n; ++i) for (int j = 1; j <= i; ++j) dp[i] = max(dp[i], price[j] + dp[i-j] - (i == j ? 0 : 1)); return dp[n]; }

int main() { vector<int> price = {0, 2, 5, 7, 8}; // price[1...n] cout << "Max value with penalty: " << rodCutPenalty(4, price) << endl; }


 All Min-Cost Parenthesizations in Matrix Chain

Print all parenthesizations that yield minimum multiplication cost.

#include <iostream> #include <vector> #include <string> #include <climits> using namespace std;

vector<vector<int>> dp, split; vector<vector<vector<string>>> paths;

void buildPaths(int i, int j) { if (i == j) { paths[i][j] = { "A" + to_string(i+1) }; return; } for (int k = split[i][j]; k <= split[i][j]; ++k) { if (paths[i][k].empty()) buildPaths(i, k); if (paths[k+1][j].empty()) buildPaths(k+1, j); for (string a : paths[i][k]) for (string b : paths[k+1][j]) paths[i][j].push_back("(" + a + b + ")"); } }

int main() { vector<int> p = {10, 20, 30, 40}; int n = p.size() - 1; dp.assign(n, vector<int>(n, INT_MAX)); split.assign(n, vector<int>(n, -1)); paths.assign(n, vector<vector<string>>(n));

for (int i = 0; i < n; ++i) dp[i][i] = 0;

for (int len = 2; len <= n; ++len)
    for (int i = 0; i <= n-len; ++i) {
        int j = i + len - 1;
        for (int k = i; k < j; ++k) {
            int cost = dp[i][k] + dp[k+1][j] + p[i]*p[k+1]*p[j+1];
            if (cost < dp[i][j]) {
                dp[i][j] = cost;
                split[i][j] = k;
            }
        }
    }

buildPaths(0, n-1);
cout << "Min cost: " << dp[0][n-1] << endl;
cout << "Parenthesizations:\n";
for (string s : paths[0][n-1]) cout << s << endl;}

 Constrained LCS (Must include a subsequence P)

#include <iostream> #include <vector> #include <string> using namespace std;

int constrainedLCS(string A, string B, string P) { int n = A.size(), m = B.size(), l = P.size(); vector<vector<vector<int>>> dp(n+1, vector<vector<int>>(m+1, vector<int>(l+1, 0)));


for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= m; ++j)
        for (int k = 0; k <= l; ++k) {
            if (A[i-1] == B[j-1]) {
                if (k < l && A[i-1] == P[k])
                    dp[i][j][k+1] = max(dp[i][j][k+1], dp[i-1][j-1][k]+1);
                else
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j-1][k]+1);
            }
            dp[i][j][k] = max({dp[i][j][k], dp[i-1][j][k], dp[i][j-1][k]});
        }

return dp[n][m][l];
}

int main() { string A = "abcde", B = "acebd", P = "bd"; cout << "Constrained LCS: " << constrainedLCS(A, B, P) << endl; }


Parallel Activity Selection on 2 Machines

#include <iostream> #include <vector> #include <algorithm> using namespace std;

struct Act { int s, e; }; bool cmp(Act a, Act b) { return a.e < b.e; }

int maxParallel(vector<Act>& A) { sort(A.begin(), A.end(), cmp); int m1 = -1, m2 = -1, count = 0; for (auto a : A) { if (a.s >= m1) { m1 = a.e; ++count; } else if (a.s >= m2) { m2 = a.e; ++count; } } return count; }

int main() { vector<Act> A = {{1,3},{2,5},{3,6},{5,8},{6,10}}; cout << "Max activities (2 machines): " << maxParallel(A) << endl; }
